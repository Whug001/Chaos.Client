# Emblems Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Players earn emblems from legend marks, first place on leaderboards and staff grants, pick one in a new Emblem tab in the profile, and show it in the world list's outer icon cell.

**Architecture:** The server loads one JSON file per emblem into an `EmblemCatalog`. It keeps each player's owned emblems in a new `emblems.json` next to `legend.json`. An `EmblemService` grants emblems at login, when a legend mark is added, and from a 30-second record sweep over the in-memory leaderboards. The world list's extras record grows from 3 to 5 bytes to carry the shown emblem's art number. Three new packets carry the emblem book. The client draws emblems from `emblNNN.spf` in `setoa.dat`, and the profile's unused Album tab becomes the Emblem page, built on the Korean `_nui_ebl` art.

**Tech Stack:** C# / .NET 10 (Chaos-Server, Chaos.Client with MonoGame), TUnit + FluentAssertions + Moq, Python 3.14 + Pillow (Unora tools), DALib file formats.

**Spec:** `docs/superpowers/specs/2026-09-26-emblems-design.md` in Chaos.Client (`main`, commits b04ead5 and e441bc9).

```csharp
// SERVER/Chaos/Services/Emblems/EmblemCatalog.cs
#region
using System.Diagnostics.CodeAnalysis;
using System.Text.Json;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;
#endregion

namespace Chaos.Services.Emblems;

/// <summary>The emblem files after validation, and any errors that skipped a file.</summary>
public sealed record EmblemCatalogLoad(IReadOnlyList<EmblemTemplate> Templates, IReadOnlyList<string> Errors);

/// <summary>
///     Every emblem, read from one JSON file each in <see cref="EmblemOptions.Directory" />, in file-name order. A bad file
///     is skipped with one logged error naming the file and the problem, so a typo never stops the server. Reload swaps
///     the whole list at once, so readers never see a half-loaded catalog.
/// </summary>
public sealed class EmblemCatalog
{
    public const int MAX_NAME_LENGTH = 28;

    private readonly IOptions<JsonSerializerOptions>? JsonOptions;
    private readonly ILogger<EmblemCatalog>? Logger;
    private readonly IOptions<EmblemOptions>? Options;
    private volatile Snapshot Current;

    public EmblemCatalog(IOptions<EmblemOptions> options, IOptions<JsonSerializerOptions> jsonOptions, ILogger<EmblemCatalog> logger)
    {
        Options = options;
        JsonOptions = jsonOptions;
        Logger = logger;
        Current = Snapshot.From([]);
        Reload();
    }

    private EmblemCatalog(IEnumerable<EmblemTemplate> templates) => Current = Snapshot.From(templates);

    /// <summary>Every valid emblem, in file-name order.</summary>
    public IReadOnlyList<EmblemTemplate> All => Current.All;

    /// <summary>A catalog with a fixed list and no folder, for tests.</summary>
    public static EmblemCatalog Fixed(IEnumerable<EmblemTemplate> templates) => new(templates);

    /// <summary>Checks and converts emblem files. Pure: the caller supplies the file names and contents.</summary>
    public static EmblemCatalogLoad Parse(IEnumerable<(string FileName, string Json)> files, JsonSerializerOptions jsonOptions)
    {
        var templates = new List<EmblemTemplate>();
        var errors = new List<string>();
        var keys = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

        foreach ((var fileName, var json) in files)
        {
            EmblemTemplateSchema? schema;

            try
            {
                schema = JsonSerializer.Deserialize<EmblemTemplateSchema>(json, jsonOptions);
            } catch (JsonException e)
            {
                errors.Add($"emblems/{fileName}: not valid JSON ({e.Message})");

                continue;
            }

            if (Validate(schema, keys) is { } problem)
            {
                errors.Add($"emblems/{fileName}: {problem}");

                continue;
            }

            var template = ToTemplate(schema!);
            keys.Add(template.Key);
            templates.Add(template);
        }

        return new EmblemCatalogLoad(templates, errors);
    }

    /// <summary>Re-reads every <c>*.json</c> in the folder. Returns the errors, which are also logged.</summary>
    public IReadOnlyList<string> Reload()
    {
        if (Options is null || JsonOptions is null)
            return [];

        var directory = Options.Value.Directory;

        var files = System.IO.Directory.Exists(directory)
            ? System.IO.Directory
                    .GetFiles(directory, "*.json", SearchOption.TopDirectoryOnly)
                    .Order(StringComparer.OrdinalIgnoreCase)
                    .Select(path => (Path.GetFileName(path), File.ReadAllText(path)))
                    .ToList()
            : [];

        var load = Parse(files, JsonOptions.Value);
        Current = Snapshot.From(load.Templates);

        foreach (var error in load.Errors)
            Logger?.LogError("Emblem file skipped: {Error}", error);

        return load.Errors;
    }

    public bool TryGet(string key, [MaybeNullWhen(false)] out EmblemTemplate template) => Current.ByKey.TryGetValue(key, out template);

    private static EmblemTemplate ToTemplate(EmblemTemplateSchema schema)
    {
        var source = schema.Source!;

        (var kind, var sourceKey) = source switch
        {
            { Staff: true }                                   => (EmblemSourceKind.Staff, (string?)null),
            { Leaderboard: { } board } when board.Length > 0 => (EmblemSourceKind.Record, CanonicalBoard(board)),
            _                                                 => (EmblemSourceKind.LegendMark, source.LegendMark)
        };

        return new EmblemTemplate
        {
            Key = schema.Key!,
            Art = (ushort)schema.Art,
            Name = schema.Name!,
            Description = schema.Description ?? string.Empty,
            Secret = schema.Secret,
            DurationDays = schema.DurationDays,
            Kind = kind,
            SourceKey = sourceKey
        };
    }

    private static string CanonicalBoard(string board)
    {
        EmblemBoardIds.All.TryGetValue(board, out var canonical);

        return canonical!;
    }

    private static string? Validate(EmblemTemplateSchema? schema, HashSet<string> keys)
    {
        if (schema is null)
            return "the file is empty";

        if (string.IsNullOrWhiteSpace(schema.Key))
            return "key is missing";

        if (keys.Contains(schema.Key))
            return $"key {schema.Key} is already used by another file";

        if (!EmblemArt.IsValid(schema.Art))
            return $"art {schema.Art} has no file";

        if (string.IsNullOrWhiteSpace(schema.Name))
            return "name is missing";

        if (schema.Name.Length > MAX_NAME_LENGTH)
            return $"name is longer than {MAX_NAME_LENGTH} characters";

        var source = schema.Source;
        var hasMark = !string.IsNullOrWhiteSpace(source?.LegendMark);
        var hasBoard = !string.IsNullOrWhiteSpace(source?.Leaderboard);
        var hasStaff = source?.Staff == true;

        if ((hasMark ? 1 : 0) + (hasBoard ? 1 : 0) + (hasStaff ? 1 : 0) != 1)
            return "source must be exactly one of legendMark, leaderboard or staff";

        if (hasBoard)
        {
            if (!EmblemBoardIds.All.Contains(source!.Leaderboard!))
                return $"leaderboard {source.Leaderboard} is not a known board";

            if (schema.DurationDays is not null)
                return "a leaderboard emblem can't have durationDays";
        }

        if (schema.DurationDays is <= 0)
            return "durationDays must be 1 or more";

        return null;
    }

    private sealed record Snapshot(IReadOnlyList<EmblemTemplate> All, IReadOnlyDictionary<string, EmblemTemplate> ByKey)
    {
        public static Snapshot From(IEnumerable<EmblemTemplate> templates)
        {
            var all = templates.ToList();

            return new Snapshot(all, all.ToDictionary(t => t.Key, StringComparer.OrdinalIgnoreCase));
        }
    }
}
```

```csharp
// SERVER/Chaos/Services/Emblems/AislingEmblems.cs
#region
using System.Diagnostics.CodeAnalysis;
#endregion

namespace Chaos.Services.Emblems;

/// <summary>When a player got an emblem, and when it runs out (null = never).</summary>
public sealed record OwnedEmblem(DateTime Granted, DateTime? Expires)
{
    public bool IsExpiredAt(DateTime nowUtc) => Expires is { } expires && (nowUtc >= expires);
}

/// <summary>
///     One player's legend-mark and staff emblems, the keys that were revoked, and the shown emblem. Record emblems are
///     not stored here; <see cref="EmblemService" /> works them out from the record holders. Every member locks, because
///     the record sweep and the player's own map can both reach a player at once.
/// </summary>
public sealed class AislingEmblems
{
    private readonly Dictionary<string, OwnedEmblem> Owned = new(StringComparer.OrdinalIgnoreCase);
    private readonly HashSet<string> Revoked = new(StringComparer.OrdinalIgnoreCase);
    private readonly Lock Sync = new();

    //the shown emblem's art and expiry, kept current by EmblemService so the world list can read them without the catalog
    private ushort ShownArt;
    private DateTime? ShownExpires;
    private string? ShownKey;

    /// <summary>The key of the emblem shown in the world list, or null.</summary>
    public string? Shown
    {
        get
        {
            lock (Sync)
                return ShownKey;
        }
    }

    /// <summary>Moves every expired emblem to the revoked keys and returns their keys. Clears the shown emblem if it expired.</summary>
    public IReadOnlyList<string> ExpireAt(DateTime nowUtc)
    {
        lock (Sync)
        {
            var expired = Owned.Where(kvp => kvp.Value.IsExpiredAt(nowUtc))
                               .Select(kvp => kvp.Key)
                               .ToList();

            foreach (var key in expired)
            {
                Owned.Remove(key);
                Revoked.Add(key);
            }

            if (ShownKey is not null && expired.Contains(ShownKey, StringComparer.OrdinalIgnoreCase))
                ClearShownLocked();

            return expired;
        }
    }

    public static AislingEmblems FromSchema(AislingEmblemsSchema schema)
    {
        var emblems = new AislingEmblems
        {
            ShownKey = string.IsNullOrWhiteSpace(schema.Shown) ? null : schema.Shown
        };

        foreach ((var key, var owned) in schema.Owned)
            emblems.Owned[key] = new OwnedEmblem(AsUtc(owned.Granted), owned.Expires is { } expires ? AsUtc(expires) : null);

        foreach (var key in schema.Revoked)
            emblems.Revoked.Add(key);

        return emblems;
    }

    /// <summary>Owns the emblem from <paramref name="grantedUtc" />, replacing any earlier grant, and un-revokes it.</summary>
    public void Grant(string key, DateTime grantedUtc, DateTime? expiresUtc)
    {
        lock (Sync)
        {
            Owned[key] = new OwnedEmblem(grantedUtc, expiresUtc);
            Revoked.Remove(key);
        }
    }

    public bool IsRevoked(string key)
    {
        lock (Sync)
            return Revoked.Contains(key);
    }

    /// <summary>Removes the emblem and marks it revoked, so its legend mark doesn't grant it again. Clears it if shown.</summary>
    public void Revoke(string key)
    {
        lock (Sync)
        {
            Owned.Remove(key);
            Revoked.Add(key);

            if (string.Equals(ShownKey, key, StringComparison.OrdinalIgnoreCase))
                ClearShownLocked();
        }
    }

    /// <summary>Sets the shown emblem and the art and expiry the world list reads. A null key clears it.</summary>
    public void SetShown(string? key, ushort art, DateTime? expiresUtc)
    {
        lock (Sync)
        {
            if (key is null)
            {
                ClearShownLocked();

                return;
            }

            ShownKey = key;
            ShownArt = art;
            ShownExpires = expiresUtc;
        }
    }

    /// <summary>The art number for the world list: the shown emblem's, or 0 when none is shown or it has expired.</summary>
    public ushort ShownArtAt(DateTime nowUtc)
    {
        lock (Sync)
            return ShownExpires is { } expires && (nowUtc >= expires) ? (ushort)0 : ShownArt;
    }

    public AislingEmblemsSchema ToSchema()
    {
        lock (Sync)
            return new AislingEmblemsSchema
            {
                Shown = ShownKey,
                Owned = Owned.ToDictionary(
                    kvp => kvp.Key,
                    kvp => new OwnedEmblemSchema
                    {
                        Granted = kvp.Value.Granted,
                        Expires = kvp.Value.Expires
                    },
                    StringComparer.OrdinalIgnoreCase),
                Revoked = Revoked.Order(StringComparer.OrdinalIgnoreCase)
                                 .ToList()
            };
    }

    public bool TryGetOwned(string key, [MaybeNullWhen(false)] out OwnedEmblem owned)
    {
        lock (Sync)
            return Owned.TryGetValue(key, out owned);
    }

    private static DateTime AsUtc(DateTime value) => value.Kind == DateTimeKind.Utc ? value : DateTime.SpecifyKind(value, DateTimeKind.Utc);

    private void ClearShownLocked()
    {
        ShownKey = null;
        ShownArt = 0;
        ShownExpires = null;
    }
}
```

```csharp
namespace Chaos.Services.Emblems;

/// <summary>One player's score on a leaderboard.</summary>
public readonly record struct EmblemBoardScore(string Name, long Score);

public static class EmblemRecordHolder
{
    /// <summary>
    ///     First place on a board. On a tie, the current holder keeps it; with no current holder among the tied, the first
    ///     name in ordinal order (ignoring case) wins, so the result is stable. Null for an empty board.
    /// </summary>
    public static string? Pick(IReadOnlyCollection<EmblemBoardScore> scores, bool lowerIsBetter, string? currentHolder)
    {
        if (scores.Count == 0)
            return null;

        var best = lowerIsBetter ? scores.Min(s => s.Score) : scores.Max(s => s.Score);

        var tied = scores.Where(s => s.Score == best)
                         .Select(s => s.Name)
                         .ToList();

        if (currentHolder is not null && tied.FirstOrDefault(name => name.EqualsI(currentHolder)) is { } kept)
            return kept;

        return tied.Order(StringComparer.OrdinalIgnoreCase)
                   .First();
    }
}
```

```csharp
/// <summary>
///     Reads the five boards from their shared <see cref="IStorage{T}" /> objects (singletons, so this sees what the scripts
///     wrote). Boards are plain dictionaries written from map threads, so a read that collides with a write is retried.
/// </summary>
public sealed class EmblemBoards(
    IStorage<PitFightLeaderboardScript.PitFightLeaderboardObj> pitFight,
    IStorage<DamageGameObj> damageDummy,
    IStorage<HopocalypseLeaderboardObj> hopocalypse,
    IStorage<HalloweenChallengeLeaderboardObj> halloween,
    IStorage<FrostyChallengeLeaderboardObj> frosty) : IEmblemBoards
{
    private const int READ_ATTEMPTS = 3;

    public bool LowerIsBetter(string boardId) => boardId.EqualsI(EmblemBoardIds.FROSTY);

    public IReadOnlyCollection<EmblemBoardScore>? Scores(string boardId)
    {
        for (var attempt = 0; attempt < READ_ATTEMPTS; attempt++)
            try
            {
                return Read(boardId)
                       .Where(score => score.Score > 0)
                       .ToList();
            } catch (InvalidOperationException)
            {
                //a script changed the board while it was being read; try again
            }

        return null;
    }

    //the damage board keys by lowercase name and stores no display name
    private static string Capitalize(string key) => key.Length == 0 ? key : char.ToUpperInvariant(key[0]) + key[1..];

    private static string NameOr(string? name, string key) => string.IsNullOrWhiteSpace(name) ? key : name;

    private IEnumerable<EmblemBoardScore> Read(string boardId)
    {
        if (boardId.EqualsI(EmblemBoardIds.PIT_FIGHT))
            return pitFight.Value.Entries.Select(kvp => new EmblemBoardScore(kvp.Key, kvp.Value.Victories));

        if (boardId.EqualsI(EmblemBoardIds.DAMAGE_DUMMY))
            return damageDummy.Value.Entries.Select(kvp => new EmblemBoardScore(Capitalize(kvp.Key), kvp.Value.Damage));

        if (boardId.EqualsI(EmblemBoardIds.HOPOCALYPSE))
            return hopocalypse.Value.Entries.Select(kvp => new EmblemBoardScore(kvp.Key, kvp.Value.HighestLevelAchieved));

        if (boardId.EqualsI(EmblemBoardIds.HALLOWEEN))
            return halloween.Value.Entries.Select(
                kvp => new EmblemBoardScore(NameOr(kvp.Value.PlayerName, kvp.Key), kvp.Value.HighestWaveAchieved));

        if (boardId.EqualsI(EmblemBoardIds.FROSTY))
            return frosty.Value.Entries.Select(kvp => new EmblemBoardScore(NameOr(kvp.Value.PlayerName, kvp.Key), kvp.Value.Seconds));

        return [];
    }
```

```csharp
// SERVER/Chaos.Networking/Converters/Server/EmblemBookConverter.cs
#region
using Chaos.IO.Memory;
using Chaos.Networking.Abstractions.Definitions;
using Chaos.Networking.Entities.Server;
using Chaos.Packets.Abstractions;
#endregion

namespace Chaos.Networking.Converters.Server;

/// <summary>Provides packet serialization and deserialization logic for <see cref="EmblemBookArgs" /></summary>
public sealed class EmblemBookConverter : PacketConverterBase<EmblemBookArgs>
{
    private const byte OWNED_FLAG = 0b_0000_0001;
    private const byte RECORD_FLAG = 0b_0000_0010;

    /// <inheritdoc />
    public override byte OpCode => (byte)ServerOpCode.EmblemBook;

    /// <inheritdoc />
    public override EmblemBookArgs Deserialize(ref SpanReader reader)
    {
        var shownKey = reader.ReadString8();
        var count = reader.ReadUInt16();
        var entries = new List<EmblemBookEntry>(count);

        for (var i = 0; i < count; i++)
        {
            var key = reader.ReadString8();
            var art = reader.ReadUInt16();
            var name = reader.ReadString8();
            var description = reader.ReadString16();
            var flags = reader.ReadByte();
            var grantedUnix = reader.ReadUInt32();
            var secondsLeft = reader.ReadUInt32();
            var holder = reader.ReadString8();

            entries.Add(
                new EmblemBookEntry
                {
                    Key = key,
                    Art = art,
                    Name = name,
                    Description = description,
                    Owned = (flags & OWNED_FLAG) != 0,
                    Record = (flags & RECORD_FLAG) != 0,
                    GrantedUnix = grantedUnix,
                    SecondsLeft = secondsLeft,
                    Holder = holder
                });
        }

        return new EmblemBookArgs
        {
            ShownKey = shownKey,
            Entries = entries
        };
    }

    /// <inheritdoc />
    public override void Serialize(ref SpanWriter writer, EmblemBookArgs args)
    {
        writer.WriteString8(args.ShownKey);
        writer.WriteUInt16((ushort)args.Entries.Count);

        foreach (var entry in args.Entries)
        {
            var flags = (byte)0;

            if (entry.Owned)
                flags |= OWNED_FLAG;

            if (entry.Record)
                flags |= RECORD_FLAG;

            writer.WriteString8(entry.Key);
            writer.WriteUInt16(entry.Art);
            writer.WriteString8(entry.Name);
            writer.WriteString16(entry.Description);
            writer.WriteByte(flags);
            writer.WriteUInt32(entry.GrantedUnix);
            writer.WriteUInt32(entry.SecondsLeft);
            writer.WriteString8(entry.Holder);
        }
    }
}
```

```csharp
// SERVER/Chaos/Services/Emblems/EmblemService.cs
#region
using Chaos.Extensions.Common;
using Chaos.Models.Legend;
using Chaos.Models.World;
using Chaos.Networking.Abstractions;
using Chaos.Networking.Entities.Server;
using Chaos.Storage.Abstractions;
using Microsoft.Extensions.Logging;
#endregion

namespace Chaos.Services.Emblems;

/// <summary>
///     Emblem rules: when emblems are granted, expire, move or are taken, which one a player shows, and what the emblem
///     book lists. Legend-mark and staff emblems live in each player's <see cref="AislingEmblems" />. Record emblems are
///     worked out live from <see cref="EmblemRecords" />, which only <see cref="SweepRecords" /> changes, so nothing is
///     ever written to an offline player's save.
/// </summary>
public sealed class EmblemService
{
    public const string EARNED = "You earned the {0} emblem.";
    public const string LOST = "You lost the {0} emblem.";

    private readonly IEmblemBoards Boards;
    private readonly EmblemCatalog Catalog;
    private readonly IClientRegistry<IChaosWorldClient> ClientRegistry;
    private readonly ILogger<EmblemService> Logger;
    private readonly IStorage<EmblemRecords> RecordStorage;
    private readonly Lock RecordSync = new();
    private readonly TimeProvider Time;

    public EmblemService(
        EmblemCatalog catalog,
        IEmblemBoards boards,
        IStorage<EmblemRecords> recordStorage,
        IClientRegistry<IChaosWorldClient> clientRegistry,
        TimeProvider time,
        ILogger<EmblemService> logger)
    {
        Catalog = catalog;
        Boards = boards;
        RecordStorage = recordStorage;
        ClientRegistry = clientRegistry;
        Time = time;
        Logger = logger;
    }

    private DateTime NowUtc => Time.GetUtcNow().UtcDateTime;

    /// <summary>
    ///     Builds the player's emblem book: owned emblems, then locked ones, each in catalog order. Unowned secret emblems
    ///     are left out.
    /// </summary>
    public EmblemBookArgs BuildBook(Aisling aisling)
    {
        var now = NowUtc;
        var owned = new List<EmblemBookEntry>();
        var locked = new List<EmblemBookEntry>();

        foreach (var template in Catalog.All)
        {
            var isOwned = Owns(aisling, template, now);

            if (!isOwned && template.Secret)
                continue;

            var isRecord = template.Kind == EmblemSourceKind.Record;

            var entry = new EmblemBookEntry
            {
                Key = template.Key,
                Art = template.Art,
                Name = template.Name,
                Description = template.Description,
                Owned = isOwned,
                Record = isRecord,
                Holder = isRecord ? HolderOf(template.Key) ?? string.Empty : string.Empty
            };

            if (isOwned && !isRecord && aisling.Emblems.TryGetOwned(template.Key, out var ownership))
            {
                entry.GrantedUnix = ToUnix(ownership.Granted);

                if (ownership.Expires is { } expires)
                    entry.SecondsLeft = (uint)Math.Max(1, Math.Ceiling((expires - now).TotalSeconds));
            }

            (isOwned ? owned : locked).Add(entry);
        }

        var shown = aisling.Emblems.Shown;

        return new EmblemBookArgs
        {
            ShownKey = shown is not null && Catalog.TryGet(shown, out var shownTemplate) && Owns(aisling, shownTemplate, now)
                ? shownTemplate.Key
                : string.Empty,
            Entries = [.. owned, .. locked]
        };
    }

    /// <summary>Staff grant. Refuses record emblems and unknown keys. Clears a revoked key and starts a fresh duration.</summary>
    public bool Give(Aisling target, string key, out string reply)
    {
        if (!TryGetGrantable(key, out var template, out reply))
            return false;

        Grant(target, template, NowUtc);
        SendBook(target);
        reply = $"You gave {target.Name} the {template.Name} emblem.";

        return true;
    }

    public void HandleBookRequest(Aisling aisling) => SendBook(aisling);

    /// <summary>The player picked an emblem to show. Empty hides. Keys the player doesn't own are ignored.</summary>
    public void HandleChoice(Aisling aisling, string key)
    {
        var now = NowUtc;

        if (string.IsNullOrEmpty(key))
            SetShown(aisling, null);
        else if (Catalog.TryGet(key, out var template) && Owns(aisling, template, now))
            SetShown(aisling, template);
        else
            return;

        SendBook(aisling);
    }

    /// <summary>Hooks the player's legend so new marks grant emblems, then runs <see cref="Refresh" />.</summary>
    public void OnLogin(Aisling aisling)
    {
        aisling.Legend.MarkAdded += mark => OnMarkAdded(aisling, mark);
        Refresh(aisling);
    }

    public void OnMarkAdded(Aisling aisling, LegendMark mark)
    {
        var now = NowUtc;
        var changed = false;

        foreach (var template in Catalog.All)
            if ((template.Kind == EmblemSourceKind.LegendMark) && template.SourceKey!.EqualsI(mark.Key) && TryAutoGrant(aisling, template, now))
                changed = true;

        if (changed)
            SendBook(aisling);
    }

    public bool Owns(Aisling aisling, EmblemTemplate template, DateTime nowUtc)
        => template.Kind == EmblemSourceKind.Record
            ? HolderOf(template.Key) is { } holder && holder.EqualsI(aisling.Name)
            : aisling.Emblems.TryGetOwned(template.Key, out var ownership) && !ownership.IsExpiredAt(nowUtc);

    /// <summary>
    ///     Moves expired emblems to revoked, grants every legend-mark emblem the player qualifies for (past achievements
    ///     count), and clears the shown emblem if the player no longer owns it. Runs at login and after /reload emblems.
    /// </summary>
    public void Refresh(Aisling aisling)
    {
        var now = NowUtc;
        var changed = aisling.Emblems.ExpireAt(now).Count > 0;

        foreach (var template in Catalog.All)
            if ((template.Kind == EmblemSourceKind.LegendMark)
                && aisling.Legend.ContainsKey(template.SourceKey!)
                && TryAutoGrant(aisling, template, now))
                changed = true;

        changed |= FixShown(aisling, now);

        if (changed)
            SendBook(aisling);
    }

    /// <summary>
    ///     Recomputes first place for every record emblem from the boards and saves any change. The old holder, if online,
    ///     loses it (and it is hidden if shown); the new holder, if online, earns it (and it shows if their cell is empty).
    /// </summary>
    public void SweepRecords()
    {
        var changes = new List<(EmblemTemplate Template, string? Old, string? New)>();

        lock (RecordSync)
        {
            var holders = RecordStorage.Value.Holders;

            foreach (var template in Catalog.All.Where(t => t.Kind == EmblemSourceKind.Record))
            {
                if (Boards.Scores(template.SourceKey!) is not { } scores)
                    continue;

                var current = holders.GetValueOrDefault(template.Key);
                var next = EmblemRecordHolder.Pick(scores, Boards.LowerIsBetter(template.SourceKey!), current);

                if (string.Equals(current, next, StringComparison.OrdinalIgnoreCase))
                    continue;

                if (next is null)
                    holders.Remove(template.Key);
                else
                    holders[template.Key] = next;

                changes.Add((template, current, next));
            }

            if (changes.Count > 0)
                RecordStorage.Save();
        }

        var now = NowUtc;

        foreach ((var template, var old, var next) in changes)
        {
            Logger.LogInformation(
                "Emblem record {Emblem} moved from {Old} to {New}",
                template.Key,
                old ?? "nobody",
                next ?? "nobody");

            if (old is not null && FindOnline(old) is { } loser)
            {
                loser.SendOrangeBarMessage(string.Format(LOST, template.Name));
                FixShown(loser, now);
                SendBook(loser);
            }

            if (next is not null && FindOnline(next) is { } winner)
            {
                winner.SendOrangeBarMessage(string.Format(EARNED, template.Name));

                if (winner.Emblems.Shown is null)
                    SetShown(winner, template);

                SendBook(winner);
            }
        }
    }

    /// <summary>Staff take. Removes the emblem and blocks its legend mark from granting it again.</summary>
    public bool Take(Aisling target, string key, out string reply)
    {
        if (!TryGetGrantable(key, out var template, out reply))
            return false;

        target.Emblems.Revoke(template.Key);
        SendBook(target);
        reply = $"You took the {template.Name} emblem from {target.Name}.";

        return true;
    }

    private Aisling? FindOnline(string name)
        => ClientRegistry.Select(client => client.Aisling)
                         .FirstOrDefault(aisling => aisling is not null && aisling.Name.EqualsI(name));

    /// <summary>Clears the shown emblem if the player no longer owns it, and refreshes the art the world list reads. True when cleared.</summary>
    private bool FixShown(Aisling aisling, DateTime now)
    {
        var shown = aisling.Emblems.Shown;

        if (shown is null)
            return false;

        if (Catalog.TryGet(shown, out var template) && Owns(aisling, template, now))
        {
            SetShown(aisling, template);

            return false;
        }

        SetShown(aisling, null);

        return true;
    }

    private void Grant(Aisling aisling, EmblemTemplate template, DateTime now)
    {
        DateTime? expires = template.DurationDays is { } days ? now.AddDays(days) : null;
        aisling.Emblems.Grant(template.Key, now, expires);
        aisling.SendOrangeBarMessage(string.Format(EARNED, template.Name));

        var shown = aisling.Emblems.Shown;

        //show it when the cell is empty; refresh the cached expiry when it is the one already shown
        if (shown is null || shown.EqualsI(template.Key))
            SetShown(aisling, template);
    }

    private string? HolderOf(string emblemKey)
    {
        lock (RecordSync)
            return RecordStorage.Value.Holders.GetValueOrDefault(emblemKey);
    }

    private void SendBook(Aisling aisling) => aisling.Client.SendEmblemBook(BuildBook(aisling));

    private void SetShown(Aisling aisling, EmblemTemplate? template)
    {
        if (template is null)
        {
            aisling.Emblems.SetShown(null, 0, null);

            return;
        }

        DateTime? expires = (template.Kind != EmblemSourceKind.Record) && aisling.Emblems.TryGetOwned(template.Key, out var ownership)
            ? ownership.Expires
            : null;

        aisling.Emblems.SetShown(template.Key, template.Art, expires);
    }

    private static uint ToUnix(DateTime utc) => (uint)new DateTimeOffset(DateTime.SpecifyKind(utc, DateTimeKind.Utc)).ToUnixTimeSeconds();

    /// <summary>Grants unless the player owns it (even expired, until the next login moves it) or it was revoked.</summary>
    private bool TryAutoGrant(Aisling aisling, EmblemTemplate template, DateTime now)
    {
        if (aisling.Emblems.IsRevoked(template.Key) || aisling.Emblems.TryGetOwned(template.Key, out _))
            return false;

        Grant(aisling, template, now);

        return true;
    }

    private bool TryGetGrantable(string key, [System.Diagnostics.CodeAnalysis.NotNullWhen(true)] out EmblemTemplate? template, out string reply)
    {
        if (!Catalog.TryGet(key, out template))
        {
            reply = $"There is no emblem called {key}.";

            return false;
        }

        if (template.Kind == EmblemSourceKind.Record)
        {
            reply = $"{template.Name} follows its leaderboard, so it can't be given or taken.";
            template = null;
```

```csharp
/// <summary>
///     Runs <see cref="EmblemService.SweepRecords" /> at startup and then every <see cref="EmblemOptions.RecordSweepSeconds" />.
///     Boards are saved from many scripts, so reading them on a timer catches every change without touching the scripts.
/// </summary>
public sealed class EmblemRecordSweepService(EmblemService service, IOptions<EmblemOptions> options, ILogger<EmblemRecordSweepService> logger)
    : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        using var timer = new PeriodicTimer(TimeSpan.FromSeconds(Math.Max(5, options.Value.RecordSweepSeconds)));

        do
            try
            {
                service.SweepRecords();
            } catch (Exception e)
            {
                logger.LogError(e, "Emblem record sweep failed");
            }
        while (await timer.WaitForNextTickAsync(stoppingToken));
    }
}
```

```python
"""Constants and pure helpers for the emblem art. See README.md."""
from __future__ import annotations

import re
from pathlib import Path

from PIL import Image

BOOK_SIZE = (599, 306)
SHIFT = 24
PANELS = (336, 58, 564, 246)       # the right page's panels below the removed top row (x0, y0, x1, y1)
BLANK_STRIP = (336, 248, 564, 272)  # blank panel texture below the description box, used to fill the freed band
BUTTON_BOX = (337, 36, 386, 55)     # the Korean book's "Name" stone button, 49x19: the base for Show/Hide
BUTTON_SIZE = (49, 19)
TAB_FRAME = 4                       # the Album tab's frame in _nui_tb1.spf (small) and _nui_tb2.spf (big)
EMBLEM_NUMBERS = [*range(1, 174), 181, 182, 183]


def emblem_file(number: int) -> str:
    return f"embl{number:03d}.spf"


def shift_right_page(page: Image.Image) -> Image.Image:
    """Removes the right page's top row: moves the panels up SHIFT px and fills the freed band with blank panel.
    The freed band (y 222-246) is 24 rows, the same height as BLANK_STRIP, so one paste fills it."""
    out = page.copy()
    out.paste(page.crop(PANELS), (PANELS[0], PANELS[1] - SHIFT))
    out.paste(page.crop(BLANK_STRIP), (PANELS[0], PANELS[3] - SHIFT))
    return out


def parse_rects(layout: str) -> dict[str, tuple[int, int, int, int]]:
    """Control name -> (left, top, right, bottom) from a prefab layout file."""
    rects = {}
    for name, rect in re.findall(r'<NAME>\s*"([^"]+)".*?<RECT>\s*([-\d ]+)', layout, re.S):
        rects[name] = tuple(int(v) for v in rect.split())
    return rects
```

```python
# UNORA/Tools/Emblems/extract_sources.py
"""One-time: copy the sources into Tools/Emblems/source and write the starting art PNGs. Run from the repo root:

    python Tools/Emblems/extract_sources.py

Existing art PNGs are never overwritten, so carved art is safe. Sources: the Korean assets on the desktop and the
client folder's setoa.dat (for the current tab frames)."""
from __future__ import annotations

import shutil
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parent / "Accessories"))
sys.path.insert(0, str(HERE.parent / "WorldList"))
sys.path.insert(0, str(HERE))

from PIL import Image  # noqa: E402

from acclib.daformats import DatArchive, Spf  # noqa: E402
from acclib.paths import DEFAULT_DATA_DIR  # noqa: E402
from emblem_art import BUTTON_BOX, EMBLEM_NUMBERS, TAB_FRAME, emblem_file, shift_right_page  # noqa: E402

ASSETS = Path.home() / "Desktop" / "Emblem Assets"


def indexed(spf: Spf, frame) -> Image.Image:
    im = Image.frombytes("P", (frame.width, frame.height), frame.data)
    im.putpalette(spf.palette_rgb())
    return im


def save_new(im: Image.Image, path: Path) -> None:
    if path.exists():
        print(f"kept {path.relative_to(HERE)}")
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    im.save(path)
    print(f"wrote {path.relative_to(HERE)}")


def main() -> int:
    src = HERE / "source"
    (src / "emblems").mkdir(parents=True, exist_ok=True)
    shutil.copy2(ASSETS / "Emblem UI" / "_nui_ebl.spf", src / "kr_nui_ebl.spf")
    for n in EMBLEM_NUMBERS:
        shutil.copy2(ASSETS / "Emblems" / emblem_file(n), src / "emblems" / emblem_file(n))
    arc = DatArchive.load(DEFAULT_DATA_DIR / "setoa.dat")
    (src / "nui_tb1.spf").write_bytes(arc["_nui_tb1.spf"])
    (src / "nui_tb2.spf").write_bytes(arc["_nui_tb2.spf"])

    book = Spf.from_bytes((src / "kr_nui_ebl.spf").read_bytes())
    page = indexed(book, book.frames[0])
    save_new(shift_right_page(page), HERE / "art" / "nui_ebl.png")
    button = page.crop(BUTTON_BOX)
    for i in range(4):
        save_new(button, HERE / "art" / "nui_eblb" / f"{i:02d}.png")
    for name in ("nui_tb1", "nui_tb2"):
        spf = Spf.from_bytes((src / f"{name}.spf").read_bytes())
        save_new(indexed(spf, spf.frames[TAB_FRAME]), HERE / "art" / f"{name}_{TAB_FRAME:02d}.png")
    print(f"copied {len(EMBLEM_NUMBERS)} emblems")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# UNORA/Tools/Emblems/build.py
"""Build the emblem book, Show/Hide button, Emblem tab and emblems into setoa.dat. Run from the repo root:

    python Tools/Emblems/build.py            # batch folder on the desktop + copy into the client folder
    python Tools/Emblems/build.py --no-apply # batch folder only

Reads art/*.png (indexed; the palette in the PNG is ignored), _nui_ebl.txt and source/, writes the files into a copy
of the client folder's setoa.dat, re-reads the copy to check every file, and writes review sheets and manifest.txt.
Upload the batch's setoa.dat with the client that uses it. Build on top of the world list drawer's setoa.dat."""
from __future__ import annotations

import argparse
import shutil
import sys
from dataclasses import replace
from datetime import datetime
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parent / "Accessories"))
sys.path.insert(0, str(HERE.parent / "WorldList"))
sys.path.insert(0, str(HERE))

from PIL import Image, ImageDraw  # noqa: E402

from acclib.daformats import DatArchive, Spf, SpfFrame  # noqa: E402
from acclib.paths import DEFAULT_DATA_DIR  # noqa: E402
from emblem_art import BOOK_SIZE, BUTTON_SIZE, EMBLEM_NUMBERS, TAB_FRAME, emblem_file  # noqa: E402
from spfpng import png_to_indexes  # noqa: E402


def build_book() -> Spf:
    spf = Spf.from_bytes((HERE / "source" / "kr_nui_ebl.spf").read_bytes())
    frame = spf.frames[0]
    frame.data = png_to_indexes(HERE / "art" / "nui_ebl.png", *BOOK_SIZE)
    return spf


def build_buttons() -> Spf:
    spf = Spf.from_bytes((HERE / "source" / "kr_nui_ebl.spf").read_bytes())
    w, h = BUTTON_SIZE
    template = spf.frames[0]
    spf.frames = [SpfFrame(0, 0, w, h, 0, 0, template.flags, w, w * h,
                           png_to_indexes(HERE / "art" / "nui_eblb" / f"{i:02d}.png", w, h)) for i in range(4)]
    return spf


def build_tab(name: str) -> Spf:
    spf = Spf.from_bytes((HERE / "source" / f"{name}.spf").read_bytes())
    frame = spf.frames[TAB_FRAME]
    spf.frames[TAB_FRAME] = replace(frame, data=png_to_indexes(HERE / "art" / f"{name}_{TAB_FRAME:02d}.png",
                                                               frame.width, frame.height))
    return spf


def files() -> dict[str, bytes]:
    out = {
        "_nui_ebl.spf": build_book().to_bytes(),
        "_nui_eblb.spf": build_buttons().to_bytes(),
        "_nui_ebl.txt": (HERE / "_nui_ebl.txt").read_bytes(),
        "_nui_tb1.spf": build_tab("nui_tb1").to_bytes(),
        "_nui_tb2.spf": build_tab("nui_tb2").to_bytes(),
    }
    for n in EMBLEM_NUMBERS:
        out[emblem_file(n)] = (HERE / "source" / "emblems" / emblem_file(n)).read_bytes()
    return out


def check(arc_bytes: bytes, expected: dict[str, bytes]) -> None:
    arc = DatArchive.from_bytes(arc_bytes)
    for name, data in expected.items():
        if arc[name] != data:
            raise SystemExit(f"{name} read back differently from its source")


def rgba(spf: Spf, frame) -> Image.Image:
    im = Image.frombytes("P", (frame.width, frame.height), frame.data)
    im.putpalette(spf.palette_rgb())
    out = im.convert("RGBA")
    out.putdata([(r, g, b, 0 if (r, g, b) == (0, 0, 0) else 255) for (r, g, b, _a) in out.getdata()])
    return out


def write_reviews(built: dict[str, bytes]) -> None:
    review = HERE / "review"
    review.mkdir(exist_ok=True)
    cols, cell = 16, 44
    sheet = Image.new("RGB", (cols * cell, ((len(EMBLEM_NUMBERS) + cols - 1) // cols) * cell), (40, 34, 28))
    draw = ImageDraw.Draw(sheet)
    for k, n in enumerate(EMBLEM_NUMBERS):
        spf = Spf.from_bytes(built[emblem_file(n)])
        icon = rgba(spf, spf.frames[0])
        icon = icon.resize((icon.width * 3, icon.height * 3), Image.NEAREST)
        x, y = (k % cols) * cell, (k // cols) * cell
        sheet.paste(icon, (x + (cell - icon.width) // 2, y + 1), icon)
        draw.text((x + 2, y + cell - 11), str(n), fill=(240, 220, 160))
    sheet.save(review / "emblems.png")
    for name, out in (("_nui_ebl.spf", "book.png"), ("_nui_eblb.spf", "buttons.png")):
        spf = Spf.from_bytes(built[name])
        frames = [rgba(spf, f) for f in spf.frames]
        strip = Image.new("RGBA", (sum(f.width for f in frames) + 4 * len(frames), max(f.height for f in frames)))
        x = 0
        for f in frames:
            strip.paste(f, (x, 0))
            x += f.width + 4
        strip.resize((strip.width * 4, strip.height * 4), Image.NEAREST).save(review / out)
    tabs = [rgba(s, s.frames[TAB_FRAME]) for s in (Spf.from_bytes(built["_nui_tb1.spf"]), Spf.from_bytes(built["_nui_tb2.spf"]))]
    strip = Image.new("RGBA", (sum(t.width for t in tabs) + 8, max(t.height for t in tabs)))
    strip.paste(tabs[0], (0, 0))
    strip.paste(tabs[1], (tabs[0].width + 8, 0))
    strip.resize((strip.width * 4, strip.height * 4), Image.NEAREST).save(review / "tabs.png")


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--data-dir", type=Path, default=DEFAULT_DATA_DIR, help="client folder to build from")
    ap.add_argument("--out", type=Path, help="output folder (default: a new folder on the desktop)")
    ap.add_argument("--no-apply", action="store_true", help="don't copy the batch into the client folder")
    args = ap.parse_args()

    built = files()
    arc = DatArchive.load(args.data_dir / "setoa.dat")
    for name, data in built.items():
        arc.put(name, data)
    data = arc.to_bytes()
    check(data, built)
    write_reviews(built)

    stamp = datetime.now()
    out = args.out or Path.home() / "Desktop" / f"Emblems batch {stamp:%Y%m%d-%H%M}"
    out.mkdir(parents=True, exist_ok=True)
    (out / "setoa.dat").write_bytes(data)
    carved = (HERE / "art" / "carved.ok").exists()
    lines = [f"Emblem batch built {stamp:%Y-%m-%d %H:%M} from {args.data_dir}", "",
             "Changed client files (upload these):",
             "  setoa.dat (_nui_ebl.spf, _nui_eblb.spf, _nui_ebl.txt, _nui_tb1.spf, _nui_tb2.spf,"
             f" {len(EMBLEM_NUMBERS)} emblNNN.spf)", ""]
    if not carved:
        lines.append("  NOT READY TO SHIP: the English labels are not carved and approved yet (art/carved.ok)")
    (out / "manifest.txt").write_text("\n".join(lines) + "\n", encoding="utf-8")

    if not args.no_apply:
        target = args.data_dir / "setoa.dat"
        shutil.copy2(target, target.with_name(f"setoa.dat.{stamp:%Y%m%d-%H%M%S}.bak"))
        shutil.copy2(out / "setoa.dat", target)
        print(f"copied the batch into {args.data_dir} (backup kept)")
    print(f"done: {out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```csharp
#region
using Chaos.Networking.Entities.Server;
#endregion

namespace Chaos.Client.Controls.World.Popups.Profile;

/// <summary>UI-free rules for the Emblem tab: paging, slot positions, and the text for time left and status.</summary>
public static class EmblemBookLayout
{
    public const int COLUMNS = 5;
    public const int PAGE_SIZE = COLUMNS * 5;
    public const int SLOT_SIZE = 36;
    public const int SLOT_STRIDE_X = 43;
    public const int SLOT_STRIDE_Y = 45;

    public static int IndexOf(IReadOnlyList<EmblemBookEntry> entries, string? key)
    {
        if (string.IsNullOrEmpty(key))
            return -1;

        for (var i = 0; i < entries.Count; i++)
            if (string.Equals(entries[i].Key, key, StringComparison.OrdinalIgnoreCase))
                return i;

        return -1;
    }

    public static int NextPage(int page, int entryCount) => (page + 1) % PageCount(entryCount);

    public static int PageCount(int entryCount) => Math.Max(1, (entryCount + PAGE_SIZE - 1) / PAGE_SIZE);

    public static string PageText(int page, int entryCount) => $"{page + 1}/{PageCount(entryCount)}";

    public static int PrevPage(int page, int entryCount)
    {
        var count = PageCount(entryCount);

        return (page - 1 + count) % count;
    }

    /// <summary>A grid slot's offset from the grid's top-left corner.</summary>
    public static (int X, int Y) SlotOffset(int slot) => (slot % COLUMNS * SLOT_STRIDE_X, slot / COLUMNS * SLOT_STRIDE_Y);

    /// <summary>The line under the description: who holds a record, when the player earned it, or Locked.</summary>
    public static string StatusText(EmblemBookEntry entry)
    {
        if (entry.Record && (entry.Holder.Length > 0))
            return $"Held by {entry.Holder}";

        if (entry.Owned && (entry.GrantedUnix > 0))
            return $"Earned {DateTimeOffset.FromUnixTimeSeconds(entry.GrantedUnix).UtcDateTime:yyyy-MM-dd}";

        return entry.Owned ? string.Empty : "Locked";
    }

    public static string TimeLeftText(EmblemBookEntry entry, long secondsLeftNow)
    {
        if (entry.Record)
            return "While first place";

        if (!entry.Owned)
            return "Locked";

        if (entry.SecondsLeft == 0)
            return "Never";

        if (secondsLeftNow <= 0)
            return "Expired";

        var left = TimeSpan.FromSeconds(secondsLeftNow);

        if (left.TotalDays >= 1)
            return $"{(int)left.TotalDays}d {left.Hours}h";

        if (left.TotalHours >= 1)
            return $"{left.Hours}h {left.Minutes}m";

        return $"{Math.Max(1, left.Minutes)}m";
    }
}
```

```csharp
#region
using Chaos.Client.Rendering;
using Microsoft.Xna.Framework.Graphics;
#endregion

namespace Chaos.Client.Controls.World.Emblems;

/// <summary>
///     Emblem art: <c>emblNNN.spf</c> in setoa.dat, by art number. Animated emblems advance one frame every
///     <see cref="FRAME_MS" /> from one shared clock, so every copy of an emblem stays in step. The textures are
///     <see cref="UiRenderer" />'s cached ones: never dispose them.
/// </summary>
public static class EmblemTextures
{
    public const int FRAME_MS = 120;

    //frame counts by art number; 0 = the file is missing. Read on the UI thread only.
    private static readonly Dictionary<ushort, int> FrameCounts = [];

    public static string FileName(ushort art) => $"embl{art:D3}.spf";

    public static int FrameAt(int frameCount, long clockMs) => frameCount <= 1 ? 0 : (int)(clockMs / FRAME_MS % frameCount);

    /// <summary>The emblem's frame for <paramref name="clockMs" />, or null for art 0 or art missing from setoa.dat.</summary>
    public static Texture2D? Get(ushort art, long clockMs)
    {
        if ((art == 0) || UiRenderer.Instance is not { } cache)
            return null;

        var fileName = FileName(art);

        if (!FrameCounts.TryGetValue(art, out var count))
            FrameCounts[art] = count = cache.GetSpfFrameCount(fileName);

        return count == 0 ? null : cache.GetSpfTexture(fileName, FrameAt(count, clockMs));
    }
}
```

```csharp
    ///     Re-binds from the current book. Keeps the page and selection while the selected key is still present;
    ///     otherwise selects the first entry on page 1.
    /// </summary>
    private void Refresh()
    {
        var entries = Book.Entries;

        if (EmblemBookLayout.IndexOf(entries, SelectedKey) < 0)
        {
            SelectedKey = entries.Count > 0 ? entries[0].Key : null;
            Page = 0;
        }

        Page = Math.Min(Page, EmblemBookLayout.PageCount(entries.Count) - 1);
        BindPage();
        BindSelection();
    }

    private EmblemBookEntry? SelectedEntry()
    {
        var index = EmblemBookLayout.IndexOf(Book.Entries, SelectedKey);

        return index >= 0 ? Book.Entries[index] : null;
    }

    private void SelectSlot(int slot)
    {
        var index = Page * EmblemBookLayout.PAGE_SIZE + slot;

        if (index >= Book.Entries.Count)
            return;

        SelectedKey = Book.Entries[index].Key;
        BindSelection();
    }

    private void TurnPage(int page)
    {
        Page = page;
        BindPage();
    }

    private void UpdateShowButton(EmblemBookEntry? entry)
    {
        if (ShowButton is null)
            return;

        ShowButton.Visible = entry is { Owned: true };

        var hide = entry is not null && entry.Key.Equals(Book.ShownKey, StringComparison.OrdinalIgnoreCase);
        var cache = UiRenderer.Instance!;

        ShowButton.NormalTexture = cache.GetSpfTexture(BUTTON_FILE, hide ? HIDE_FRAME : SHOW_FRAME);
        ShowButton.PressedTexture = cache.GetSpfTexture(BUTTON_FILE, hide ? HIDE_PRESSED_FRAME : SHOW_PRESSED_FRAME);
    }

    private void UpdateTimeLeft()
    {
        var entry = SelectedEntry();
        var text = entry is null ? string.Empty : EmblemBookLayout.TimeLeftText(entry, Book.SecondsLeftAt(entry, DateTime.UtcNow));

        if (TimeLabel.Text != text)
            TimeLabel.Text = text;
    }
}
```

`UIButton.Dispose` disposes its Normal, Pressed, Selected and Hover textures. The Show/Hide textures are UiRenderer's shared cached ones, so `Dispose` above clears them first.
