# Quest Log Window Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a quest log window to the client, built from the Legend of Darkness "Quest List" art, which lists every active quest with live progress, lets safe quests be given up, and opens from a "Q" HUD button.

**Architecture:** One JSON entry per quest in Unora (`Data/Configuration/Templates/QuestLog/`) says which saved stage enum or flag makes the quest active, what to show at each stage, what to count, and what give-up clears. The server loads the entries once (`QuestLogCatalog`). `QuestLogService` builds each player's list from their saved trackers and inventory, and sends it on open, on change (checked on the player's one-second tick) and after a give-up. Three new packets carry the window (`QuestLogRequest`, `QuestLogDisplay`, `HotkeyHelpOpen`). The client shows the list in `QuestLogControl`, which draws the LoD art from setoa.dat and keeps its state in `WorldState.QuestLog`.

**Tech Stack:** C# / .NET 10 (Chaos-Server, and Chaos.Client with MonoGame), System.Text.Json, TUnit + FluentAssertions + Moq, Python 3 + Pillow + pytest (Unora tools), DALib file formats.

**Spec:** `docs/superpowers/specs/2026-09-27-quest-log-design.md` in Chaos.Client (`main`). Read its **Amendments** section: it overrides the spec body where the two differ.

```csharp
    private static readonly Encoding Enc = Encoding.GetEncoding(949);

    private static byte[] Write<T>(PacketConverterBase<T> converter, T args) where T: class, IPacketSerializable
    {
        var writer = new SpanWriter(Enc, usePooling: false);
        converter.Serialize(ref writer, args);

        return writer.ToSpan()
                     .ToArray();
    }

    private static T RoundTrip<T>(PacketConverterBase<T> converter, T original) where T: class, IPacketSerializable
    {
        var bytes = Write(converter, original);
        var reader = new SpanReader(Enc, bytes);

        return converter.Deserialize(ref reader);
    }
```

```csharp
using Chaos.DarkAges.Definitions;
using Chaos.IO.Memory;
using Chaos.Networking.Abstractions.Definitions;
using Chaos.Networking.Entities.Client;
using Chaos.Packets.Abstractions;

namespace Chaos.Networking.Converters.Client;

/// <summary>
///     Provides packet serialization and deserialization logic for <see cref="QuestLogRequestArgs" />
/// </summary>
public sealed class QuestLogRequestConverter : PacketConverterBase<QuestLogRequestArgs>
{
    /// <inheritdoc />
    public override byte OpCode => (byte)ClientOpCode.QuestLogRequest;

    /// <inheritdoc />
    public override QuestLogRequestArgs Deserialize(ref SpanReader reader)
    {
        var type = (QuestLogRequestType)reader.ReadByte();

        if (!Enum.IsDefined(type))
            throw new ArgumentOutOfRangeException(nameof(reader), type, "Unknown quest log request type");

        return new QuestLogRequestArgs
        {
            Type = type,
            QuestKey = type == QuestLogRequestType.GiveUp ? reader.ReadString8() : null
        };
    }

    /// <inheritdoc />
    public override void Serialize(ref SpanWriter writer, QuestLogRequestArgs args)
    {
        writer.WriteByte((byte)args.Type);

        if (args.Type == QuestLogRequestType.GiveUp)
            writer.WriteString8(args.QuestKey ?? string.Empty);
    }
}
```

`SERVER/Chaos.Networking/Converters/Server/QuestLogDisplayConverter.cs`:

```csharp
using Chaos.IO.Memory;
using Chaos.Networking.Abstractions.Definitions;
using Chaos.Networking.Entities.Server;
using Chaos.Packets.Abstractions;

namespace Chaos.Networking.Converters.Server;

/// <summary>
///     Provides packet serialization and deserialization logic for <see cref="QuestLogDisplayArgs" />
/// </summary>
public sealed class QuestLogDisplayConverter : PacketConverterBase<QuestLogDisplayArgs>
{
    /// <inheritdoc />
    public override byte OpCode => (byte)ServerOpCode.QuestLogDisplay;

    /// <inheritdoc />
    public override QuestLogDisplayArgs Deserialize(ref SpanReader reader)
    {
        var count = reader.ReadUInt16();
        var entries = new List<QuestLogEntryInfo>(count);

        for (var i = 0; i < count; i++)
        {
            var key = reader.ReadString8();
            var title = reader.ReadString8();
            var area = reader.ReadString8();
            var givenBy = reader.ReadString8();
            var text = reader.ReadString16();
            var canGiveUp = reader.ReadBoolean();
            var progressCount = reader.ReadByte();
            var progress = new List<QuestLogProgressInfo>(progressCount);

            for (var j = 0; j < progressCount; j++)
                progress.Add(new QuestLogProgressInfo(reader.ReadString8(), reader.ReadUInt16(), reader.ReadUInt16()));

            entries.Add(
                new QuestLogEntryInfo
                {
                    Key = key,
                    Title = title,
                    Area = area,
                    GivenBy = givenBy,
                    Text = text,
                    CanGiveUp = canGiveUp,
                    Progress = progress
                });
        }

        return new QuestLogDisplayArgs { Entries = entries };
    }

    /// <inheritdoc />
    public override void Serialize(ref SpanWriter writer, QuestLogDisplayArgs args)
    {
        writer.WriteUInt16((ushort)args.Entries.Count);

        foreach (var entry in args.Entries)
        {
            writer.WriteString8(entry.Key);
            writer.WriteString8(entry.Title);
            writer.WriteString8(entry.Area);
            writer.WriteString8(entry.GivenBy);
            writer.WriteString16(entry.Text);
            writer.WriteBoolean(entry.CanGiveUp);
            writer.WriteByte((byte)entry.Progress.Count);

            foreach (var line in entry.Progress)
            {
                writer.WriteString8(line.Label);
                writer.WriteUInt16(line.Have);
                writer.WriteUInt16(line.Need);
            }
        }
    }
}
```

```csharp
#region
using System.Text.Json;
using Chaos.DarkAges.Definitions;
#endregion

namespace Chaos.Services.QuestLog;

/// <summary>
///     Loads the quest log entry files. A bad file is skipped with a warning naming the file and the reason, so one typo
///     never stops the server. <c>QuestLogRealDataTests</c> loads the real Unora folder and fails on any skipped file.
/// </summary>
public static class QuestLogCatalogLoader
{
    public const int MAX_KEY_LENGTH = 40;
    public const int MAX_SHORT_TEXT = 60;
    public const int MAX_TEXT = 2000;
    public const int MAX_PROGRESS_LINES = 8;

    private static readonly JsonSerializerOptions JsonOptions = new()
    {
        PropertyNameCaseInsensitive = true,
        ReadCommentHandling = JsonCommentHandling.Skip,
        AllowTrailingCommas = true
    };

    //simple enum name -> type; null when the name exists in both namespaces
    private static readonly Lazy<IReadOnlyDictionary<string, Type?>> EnumIndex = new(BuildEnumIndex);

    public static QuestLogCatalog Load(string directory, QuestLogItemIndex items, ILogger logger)
    {
        var quests = new List<QuestDefinition>();
        var problems = new List<string>();
        var keys = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

        if (!Directory.Exists(directory))
            problems.Add($"{directory}: folder not found");
        else
            foreach (var path in Directory.EnumerateFiles(directory, "*.json", SearchOption.AllDirectories)
                                          .Order(StringComparer.OrdinalIgnoreCase))
            {
                var name = Path.GetRelativePath(directory, path)
                               .Replace('\\', '/');

                try
                {
                    var file = JsonSerializer.Deserialize<QuestLogFile>(File.ReadAllText(path), JsonOptions)
                               ?? throw new QuestLogEntryException("the file is empty");

                    var quest = Resolve(file, items);

                    if (!keys.Add(quest.Key))
                        throw new QuestLogEntryException($"key '{quest.Key}' is already used by another entry");

                    quests.Add(quest);
                } catch (JsonException ex)
                {
                    problems.Add($"{name}: invalid JSON ({ex.Message})");
                } catch (QuestLogEntryException ex)
                {
                    problems.Add($"{name}: {ex.Message}");
                }
            }

        foreach (var problem in problems)
            logger.LogWarning("Quest log entry skipped: {Problem}", problem);

        logger.LogInformation("Loaded {Count} quest log entries", quests.Count);

        return new QuestLogCatalog(quests, problems);
    }

    private static QuestDefinition Resolve(QuestLogFile file, QuestLogItemIndex items)
    {
        RequireText(file.Key, "key", MAX_KEY_LENGTH);
        RequireText(file.Title, "title", MAX_SHORT_TEXT);
        RequireText(file.Area, "area", MAX_SHORT_TEXT);

        if (file.GivenBy is { Length: > MAX_SHORT_TEXT })
            throw new QuestLogEntryException($"givenBy is longer than {MAX_SHORT_TEXT} characters");

        Type stageType;
        Enum? flagValue = null;

        switch (Present(file.StageEnum), Present(file.StageFlag))
        {
            case ({ } enumName, null):
                stageType = ResolveEnum(enumName, "stageEnum");

                break;
            case (null, { } flagName):
                flagValue = ResolveFlag(flagName, "stageFlag");
                stageType = flagValue.GetType();

                break;
            default:
                throw new QuestLogEntryException("set exactly one of stageEnum and stageFlag");
        }

        if (file.Stages.Count == 0)
            throw new QuestLogEntryException("stages is empty");

        var stages = new Dictionary<string, QuestStageDefinition>(StringComparer.Ordinal);

        foreach (var (stageName, stage) in file.Stages)
        {
            if (flagValue is null)
            {
                if (!Enum.GetNames(stageType).Contains(stageName, StringComparer.Ordinal))
                    throw new QuestLogEntryException($"stage '{stageName}' is not a member of {stageType.Name}");
            } else if (stageName != QuestDefinition.FLAG_STAGE)
                throw new QuestLogEntryException($"a stageFlag entry has one stage, named '{QuestDefinition.FLAG_STAGE}', not '{stageName}'");

            RequireText(stage.Text, $"stages.{stageName}.text", MAX_TEXT);

            if (stage.Progress.Count > MAX_PROGRESS_LINES)
                throw new QuestLogEntryException($"stages.{stageName} has more than {MAX_PROGRESS_LINES} progress lines");

            stages[stageName] = new QuestStageDefinition(
                stage.Text,
                stage.Progress
                     .Select(line => ResolveProgress(line, stageName, items))
                     .ToList());
        }

        return new QuestDefinition(
            file.Key,
            file.Title,
            file.Area,
            file.GivenBy ?? string.Empty,
            stageType,
            flagValue,
            stages,
            ResolveGiveUp(file.GiveUp));
    }

    private static QuestProgressDefinition ResolveProgress(QuestLogProgressFile line, string stageName, QuestLogItemIndex items)
    {
        var where = $"stages.{stageName}.progress";

        RequireText(line.Label, $"{where}.label", MAX_SHORT_TEXT);

        if (line.Need is < 1 or > ushort.MaxValue)
            throw new QuestLogEntryException($"{where} '{line.Label}': need must be 1 to {ushort.MaxValue}");

        var item = Present(line.Item);
        var itemName = Present(line.ItemName);
        var counter = Present(line.Counter);

        if (new[] { item, itemName, counter }.Count(value => value is not null) != 1)
            throw new QuestLogEntryException($"{where} '{line.Label}': set exactly one of item, itemName and counter");

        if (item is not null)
        {
            if (!items.TemplateKeys.Contains(item))
                throw new QuestLogEntryException($"{where} '{line.Label}': item '{item}' is not an item template key");

            return new QuestProgressDefinition(line.Label, QuestProgressKind.ItemTemplateKey, item, line.Need);
        }

        if (itemName is not null)
        {
            if (!items.Names.Contains(itemName))
                throw new QuestLogEntryException($"{where} '{line.Label}': itemName '{itemName}' is not an item template's name");

            return new QuestProgressDefinition(line.Label, QuestProgressKind.ItemName, itemName, line.Need);
        }

        RequireText(counter, $"{where} '{line.Label}' counter", MAX_SHORT_TEXT);

        return new QuestProgressDefinition(line.Label, QuestProgressKind.Counter, counter!, line.Need);
    }

    private static QuestGiveUpDefinition? ResolveGiveUp(QuestLogGiveUpFile? giveUp)
    {
        if (giveUp is null)
            return null;

        var definition = new QuestGiveUpDefinition(
            giveUp.RemoveEnums
                  .Select(name => ResolveEnum(name, "giveUp.removeEnums"))
                  .ToList(),
            giveUp.RemoveCounters
                  .Select(key =>
                  {
                      RequireText(key, "giveUp.removeCounters", MAX_SHORT_TEXT);

                      return key;
                  })
                  .ToList(),
            giveUp.RemoveFlags
                  .Select(flag => ResolveFlag(flag, "giveUp.removeFlags"))
                  .ToList());

        if (definition is { Enums.Count: 0, Counters.Count: 0, Flags.Count: 0 })
            throw new QuestLogEntryException("giveUp clears nothing; leave it out if the quest can't be given up");

        return definition;
    }

    private static Type ResolveEnum(string name, string field)
    {
        if (!EnumIndex.Value.TryGetValue(name, out var type))
            throw new QuestLogEntryException($"{field}: '{name}' is not an enum in Chaos.Definitions or Chaos.DarkAges.Definitions");

        return type ?? throw new QuestLogEntryException($"{field}: '{name}' is in both Chaos.Definitions and Chaos.DarkAges.Definitions");
    }

    private static Enum ResolveFlag(string text, string field)
    {
        var dot = text.LastIndexOf('.');

        if ((dot <= 0) || (dot == text.Length - 1))
            throw new QuestLogEntryException($"{field}: '{text}' is not written as EnumName.Member");

        var type = ResolveEnum(text[..dot], field);
        var member = text[(dot + 1)..];

        if (!type.IsDefined(typeof(FlagsAttribute), false))
            throw new QuestLogEntryException($"{field}: {type.Name} is not a [Flags] enum");

        if (!Enum.GetNames(type).Contains(member, StringComparer.Ordinal))
            throw new QuestLogEntryException($"{field}: '{member}' is not a member of {type.Name}");

        return (Enum)Enum.Parse(type, member);
    }

    private static void RequireText(string? value, string field, int maxLength)
    {
        if (string.IsNullOrWhiteSpace(value))
            throw new QuestLogEntryException($"{field} is empty");

        if (value.Length > maxLength)
            throw new QuestLogEntryException($"{field} is longer than {maxLength} characters");
    }

    private static string? Present(string? value) => string.IsNullOrWhiteSpace(value) ? null : value;

    private static IReadOnlyDictionary<string, Type?> BuildEnumIndex()
    {
        var index = new Dictionary<string, Type?>(StringComparer.Ordinal);

        var types = typeof(QuestLogCatalogLoader).Assembly
                                                 .GetTypes()
                                                 .Concat(typeof(QuestLogRequestType).Assembly.GetTypes())
```

```csharp
#region
using System.Runtime.CompilerServices;
using Chaos.Extensions.Common;
using Chaos.Models.World;
using Chaos.Networking.Entities.Server;
#endregion

namespace Chaos.Services.QuestLog;

/// <summary>
///     Builds each player's quest log from their saved trackers and inventory, and keeps an open window current. The
///     aisling script calls <see cref="Tick" /> once a second; the list is sent only when it changed. Open windows are
///     held weakly, so a player who logs out is forgotten without a logout hook.
/// </summary>
public sealed class QuestLogService(QuestLogCatalog catalog, ILogger<QuestLogService> logger)
{
    private const string MAIN_STORY = "Main Story";

    //shared so that "with { Progress = NoProgress }" compares equal in SameEntries
    private static readonly List<QuestLogProgressInfo> NoProgress = [];

    private readonly ConditionalWeakTable<Aisling, OpenWindow> OpenWindows = new();

    /// <summary>The player's active quests, in display order.</summary>
    public IReadOnlyList<QuestLogEntryInfo> Build(Aisling aisling)
    {
        var entries = new List<QuestLogEntryInfo>();

        foreach (var quest in catalog.Quests)
        {
            if ((CurrentStage(aisling, quest) is not { } stageName) || !quest.Stages.TryGetValue(stageName, out var stage))
                continue;

            entries.Add(
                new QuestLogEntryInfo
                {
                    Key = quest.Key,
                    Title = quest.Title,
                    Area = quest.Area,
                    GivenBy = quest.GivenBy,
                    Text = stage.Text,
                    CanGiveUp = quest.CanGiveUp,
                    Progress = stage.Progress
                                    .Select(line => new QuestLogProgressInfo(
                                        line.Label,
                                        (ushort)Math.Min(line.Need, Count(aisling, line)),
                                        (ushort)line.Need))
                                    .ToList()
                });
        }

        return entries.OrderBy(entry => entry.Area.EqualsI(MAIN_STORY) ? 0 : 1)
                      .ThenBy(entry => entry.Area, StringComparer.OrdinalIgnoreCase)
                      .ThenBy(entry => entry.Title, StringComparer.OrdinalIgnoreCase)
                      .ToList();
    }

    public void Open(Aisling aisling)
    {
        var entries = Build(aisling);
        OpenWindows.AddOrUpdate(aisling, new OpenWindow(entries));
        aisling.Client.SendQuestLog(entries);
    }

    public void Close(Aisling aisling) => OpenWindows.Remove(aisling);

    public bool IsOpen(Aisling aisling) => OpenWindows.TryGetValue(aisling, out _);

    /// <summary>Called once a second per player. Sends the list only when the window is open and the list changed.</summary>
    public void Tick(Aisling aisling)
    {
        if (!OpenWindows.TryGetValue(aisling, out var window))
            return;

        var entries = Build(aisling);

        if (SameEntries(window.LastSent, entries))
            return;

        window.LastSent = entries;
        aisling.Client.SendQuestLog(entries);
    }

    /// <summary>
    ///     Abandons a quest the player has active, if its entry says it is safe. Clears only what the entry lists and
    ///     never timed events, so a cooldown started on accept keeps running.
    /// </summary>
    public void GiveUp(Aisling aisling, string? questKey)
    {
        var quest = string.IsNullOrEmpty(questKey) ? null : catalog.Get(questKey);

        if (quest?.GiveUp is not { } giveUp || (CurrentStage(aisling, quest) is not { } stage) || !quest.Stages.ContainsKey(stage))
        {
            logger.LogDebug("Ignored a quest log give up of {QuestKey} from {Aisling}", questKey, aisling.Name);

            return;
        }

        foreach (var type in giveUp.Enums)
            aisling.Trackers.Enums.Remove(type);

        foreach (var key in giveUp.Counters)
            aisling.Trackers.Counters.Remove(key, out _);

        foreach (var flag in giveUp.Flags)
            aisling.Trackers.Flags.RemoveFlag(flag.GetType(), flag);

        aisling.SendOrangeBarMessage($"You gave up on {quest.Title}.");

        var entries = Build(aisling);

        if (OpenWindows.TryGetValue(aisling, out var window))
            window.LastSent = entries;

        aisling.Client.SendQuestLog(entries);
    }

    private static string? CurrentStage(Aisling aisling, QuestDefinition quest)
    {
        if (quest.FlagValue is { } flag)
            return aisling.Trackers.Flags.HasFlag(quest.StageType, flag) ? QuestDefinition.FLAG_STAGE : null;

        return aisling.Trackers.Enums.TryGetValue(quest.StageType, out var value) ? value.ToString() : null;
    }

    private static int Count(Aisling aisling, QuestProgressDefinition line)
        => line.Kind switch
        {
            QuestProgressKind.ItemTemplateKey => aisling.Inventory.CountOfByTemplateKey(line.Key),
            QuestProgressKind.ItemName        => aisling.Inventory.CountOf(line.Key),
            _                                 => aisling.Trackers.Counters.TryGetValue(line.Key, out var value) ? Math.Max(0, value) : 0
        };

    internal static bool SameEntries(IReadOnlyList<QuestLogEntryInfo> a, IReadOnlyList<QuestLogEntryInfo> b)
        => (a.Count == b.Count)
           && a.Zip(b)
               .All(pair => ((pair.First with { Progress = NoProgress }) == (pair.Second with { Progress = NoProgress }))
                            && pair.First.Progress.SequenceEqual(pair.Second.Progress));

    private sealed class OpenWindow(IReadOnlyList<QuestLogEntryInfo> lastSent)
    {
        public IReadOnlyList<QuestLogEntryInfo> LastSent { get; set; } = lastSent;
    }
}
```

```python
"""Quest log entry review page and inventory check. Run from the repo root:

    python Tools/QuestLog/review.py            # writes "Quest log review.html" on the desktop and opens it
    python Tools/QuestLog/review.py --check    # exit 1 if entries and Tools/QuestLog/inventory.md disagree

Entries are Data/Configuration/Templates/QuestLog/**/*.json. The inventory is a markdown table with the columns
key | area | batch | stage source | scripts | quest | notes, where quest is yes or no."""
from __future__ import annotations

import argparse
import html
import json
import os
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
REPO = HERE.parent.parent
ENTRIES = REPO / "Data" / "Configuration" / "Templates" / "QuestLog"
INVENTORY = HERE / "inventory.md"
COLUMNS = ["key", "area", "batch", "stage source", "scripts", "quest", "notes"]


def parse_inventory(path: Path) -> list[dict[str, str]]:
    rows = []
    for line in path.read_text(encoding="utf-8").splitlines():
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        if len(cells) != len(COLUMNS) or cells[0] in ("key", "") or set(cells[0]) <= {"-"}:
            continue
        rows.append(dict(zip(COLUMNS, cells)))
    return rows


def load_entries(folder: Path) -> list[dict]:
    entries = []
    for path in sorted(folder.rglob("*.json")):
        data = json.loads(path.read_text(encoding="utf-8"))
        data["_file"] = path.relative_to(folder).as_posix()
        entries.append(data)
    return entries


def check(inventory: list[dict[str, str]], entries: list[dict]) -> list[str]:
    entry_keys = {e["key"].lower() for e in entries}
    inventory_keys = {r["key"].lower() for r in inventory}
    problems = [f"{r['key']}: marked yes in the inventory but has no entry file"
                for r in inventory if r["quest"].lower() == "yes" and r["key"].lower() not in entry_keys]
    problems += [f"{e['key']}: has an entry file but no inventory row"
                 for e in entries if e["key"].lower() not in inventory_keys]
    return problems


def _progress(line: dict) -> str:
    source = next(f"{kind} {line[field]}" for field, kind in
                  (("item", "item"), ("itemName", "item name"), ("counter", "counter")) if line.get(field))
    return f"{line['label']}: need {line['need']} ({source})"


def _give_up(entry: dict) -> str:
    give_up = entry.get("giveUp")
    if not give_up:
        return "Cannot be given up"
    parts = [f"{name} {', '.join(give_up[field])}" for field, name in
             (("removeEnums", "enums"), ("removeCounters", "counters"), ("removeFlags", "flags")) if give_up.get(field)]
    return "Clears: " + "; ".join(parts)


def render(entries: list[dict], inventory: list[dict[str, str]]) -> str:
    esc = lambda s: html.escape(str(s), quote=True)  # noqa: E731
    order = sorted(entries, key=lambda e: (e["area"].lower() != "main story", e["area"].lower(), e["title"].lower()))
    out = ["<!doctype html><meta charset='utf-8'><title>Quest log review</title>",
           "<style>body{font:14px system-ui;max-width:900px;margin:24px auto;padding:0 16px}"
           ".q{border:1px solid #ccc;border-radius:6px;padding:10px 14px;margin:10px 0}"
           ".k{color:#777;font-size:12px}.s{margin:6px 0 6px 12px}.g{color:#a33}.ok{color:#262}</style>",
           f"<h1>Quest log review</h1><p>{len(entries)} entries.</p>"]
    area = None
    for e in order:
        if e["area"] != area:
            area = e["area"]
            out.append(f"<h2>{esc(area)}</h2>")
        source = e.get("stageEnum") or e.get("stageFlag")
        out.append(f"<div class='q'><b>{esc(e['title'])}</b> <span class='k'>{esc(e['key'])} · {esc(e['_file'])} · "
                   f"{esc(source)}</span>")
        if e.get("givenBy"):
            out.append(f"<div>Given by: {esc(e['givenBy'])}</div>")
        for stage, body in e["stages"].items():
            lines = "".join(f"<li>{esc(_progress(p))}</li>" for p in body.get("progress", []))
            out.append(f"<div class='s'><b>{esc(stage)}</b>: {esc(body['text'])}<ul>{lines}</ul></div>")
        give_up = _give_up(e)
        out.append(f"<div class='{'g' if give_up.startswith('Cannot') else 'ok'}'>{esc(give_up)}</div>")
        if e.get("notes"):
            out.append(f"<div class='k'>Notes: {esc(e['notes'])}</div>")
        out.append("</div>")
    excluded = [r for r in inventory if r["quest"].lower() != "yes"]
    out.append(f"<h2>Left out ({len(excluded)})</h2><ul>")
    out += [f"<li><b>{esc(r['key'])}</b> ({esc(r['scripts'])}): {esc(r['notes'])}</li>" for r in excluded]
    out.append("</ul>")
    return "\n".join(out)


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--check", action="store_true", help="only compare entries with the inventory")
    ap.add_argument("--out", type=Path, default=Path.home() / "Desktop" / "Quest log review.html")
    args = ap.parse_args()

    inventory = parse_inventory(INVENTORY)
    entries = load_entries(ENTRIES)
    problems = check(inventory, entries)
    for problem in problems:
        print(problem)
    if args.check:
        print(f"{len(entries)} entries, {len(problems)} problems")
        return 1 if problems else 0

    args.out.write_text(render(entries, inventory), encoding="utf-8")
    print(f"wrote {args.out}")
    if sys.platform == "win32":
        os.startfile(args.out)  # noqa: S606
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```csharp
using System.Text;
using Chaos.Networking.Entities.Server;

namespace Chaos.Client.ViewModel;

/// <summary>
///     What the quest log window shows: the server's list of active quests, the selected quest (kept by key across
///     updates) and the GiveUp confirm. The server sends the whole list on open and whenever it changes.
/// </summary>
public sealed class QuestLogState
{
    public const float CONFIRM_SECONDS = 5f;

    private readonly List<QuestLogEntryInfo> EntryList = [];
    private float ConfirmSecondsLeft;

    /// <summary>The same list instance for the life of the state, so a row list can bind it once.</summary>
    public IReadOnlyList<QuestLogEntryInfo> Entries => EntryList;

    public string? SelectedKey { get; private set; }

    /// <summary>The quest whose GiveUp is armed, waiting for the second click.</summary>
    public string? ConfirmingKey { get; private set; }

    public QuestLogEntryInfo? Selected => EntryList.FirstOrDefault(entry => entry.Key == SelectedKey);

    public bool CanGiveUp => Selected?.CanGiveUp == true;

    /// <summary>The list, the selection or the confirm changed.</summary>
    public event Action? Changed;

    public void Apply(IReadOnlyList<QuestLogEntryInfo> entries)
    {
        EntryList.Clear();
        EntryList.AddRange(entries);

        if (Selected is null)
            SelectedKey = EntryList.Count > 0 ? EntryList[0].Key : null;

        if ((ConfirmingKey is not null) && EntryList.All(entry => entry.Key != ConfirmingKey))
            ClearConfirm();

        Changed?.Invoke();
    }

    public void Select(string key)
    {
        if (EntryList.All(entry => entry.Key != key))
            return;

        SelectedKey = key;
        ClearConfirm();
        Changed?.Invoke();
    }

    /// <summary>The GiveUp button. The first click arms the confirm; the second, within 5 s, returns the key to send.</summary>
    public string? PressGiveUp()
    {
        if (!CanGiveUp)
            return null;

        if (ConfirmingKey == SelectedKey)
        {
            var key = ConfirmingKey;
            ClearConfirm();
            Changed?.Invoke();

            return key;
        }

        ConfirmingKey = SelectedKey;
        ConfirmSecondsLeft = CONFIRM_SECONDS;
        Changed?.Invoke();

        return null;
    }

    public void CancelGiveUp()
    {
        if (ConfirmingKey is null)
            return;

        ClearConfirm();
        Changed?.Invoke();
    }

    /// <summary>Ticked by the world screen, in seconds. Ends an armed confirm after 5 s.</summary>
    public void Update(float seconds)
    {
        if (ConfirmingKey is null)
            return;

        ConfirmSecondsLeft -= seconds;

        if (ConfirmSecondsLeft <= 0f)
            CancelGiveUp();
    }

    public void Clear()
    {
        EntryList.Clear();
        SelectedKey = null;
        ClearConfirm();
        Changed?.Invoke();
    }

    /// <summary>The details pane's text, with {=x} colour codes: gold title, grey giver, white text, lime when done.</summary>
    public string DetailsText()
    {
        if (Selected is not { } quest)
            return "{=i}You have no active quests.";

        if (ConfirmingKey == quest.Key)
            return $"{{=u}}Click GiveUp again to abandon {quest.Title}.";

        var text = new StringBuilder();
        text.Append("{=c}").Append(quest.Title);

        if (!string.IsNullOrEmpty(quest.GivenBy))
            text.Append("\n{=i}").Append(quest.GivenBy);

        text.Append("\n\n{=u}").Append(quest.Text);

        if (quest.Progress.Count > 0)
        {
            text.Append('\n');

            foreach (var line in quest.Progress)
                text.Append("\n{=")
                    .Append(line.Have >= line.Need ? 'q' : 'u')
                    .Append('}')
                    .Append(line.Label)
                    .Append(": ")
                    .Append(line.Have)
                    .Append(" / ")
                    .Append(line.Need);
        }

        return text.ToString();
    }

    private void ClearConfirm()
    {
        ConfirmingKey = null;
        ConfirmSecondsLeft = 0f;
    }
}
```

- [ ] **Step 3: Write the build script**

`UNORA/Tools/QuestLog/build.py`:

```python
"""Pack the quest log window's sprites into setoa.dat. Run from the repo root:

    python Tools/QuestLog/build.py            # batch folder on the desktop + copy into the client folder
    python Tools/QuestLog/build.py --no-apply # batch folder only

The sprites are the Legend of Darkness "Quest List" window, kept as-is in source/. Writes them into a copy of the
client folder's setoa.dat, re-reads the copy to check every file, and writes manifest.txt. Upload the batch's
setoa.dat in the same launcher patch as the client that draws the quest log."""
from __future__ import annotations

import argparse
import shutil
import sys
from datetime import datetime
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parent / "Accessories"))

from acclib.daformats import DatArchive  # noqa: E402
from acclib.paths import DEFAULT_DATA_DIR  # noqa: E402

SOURCE = HERE / "source"

#sprite file -> (frame count, (width, height))
FILES: dict[str, tuple[int, tuple[int, int]]] = {
    "q_list.spf": (1, (247, 307)),
    "q_min.spf": (1, (247, 32)),
    "q_close.spf": (2, (50, 20)),
    "q_gvp.spf": (2, (50, 20)),
    "under_bn.spf": (2, (18, 18)),
    "full_bn.spf": (2, (18, 18)),
    "x_bn.spf": (2, (18, 18)),
    "q_btm.spf": (2, (18, 18)),
}


def files() -> dict[str, bytes]:
    return {name: (SOURCE / name).read_bytes() for name in FILES}


def check(arc_bytes: bytes, expected: dict[str, bytes]) -> None:
    arc = DatArchive.from_bytes(arc_bytes)
    for name, data in expected.items():
        if arc[name] != data:
            raise SystemExit(f"{name} read back differently from its source")


def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--data-dir", type=Path, default=DEFAULT_DATA_DIR, help="client folder to build from")
    ap.add_argument("--out", type=Path, help="output folder (default: a new folder on the desktop)")
    ap.add_argument("--no-apply", action="store_true", help="don't copy the batch into the client folder")
    args = ap.parse_args()

    built = files()
    arc = DatArchive.load(args.data_dir / "setoa.dat")
    for name, data in built.items():
        arc.put(name, data)
    data = arc.to_bytes()
    check(data, built)

    stamp = datetime.now()
    out = args.out or Path.home() / "Desktop" / f"Quest log art batch {stamp:%Y%m%d-%H%M}"
    out.mkdir(parents=True, exist_ok=True)
    (out / "setoa.dat").write_bytes(data)
    lines = [f"Quest log art batch built {stamp:%Y-%m-%d %H:%M} from {args.data_dir}", "",
             "Changed client files (upload these, in the same launcher patch as the client with the quest log):",
             f"  setoa.dat ({', '.join(FILES)})", ""]
    (out / "manifest.txt").write_text("\n".join(lines) + "\n", encoding="utf-8")

    if not args.no_apply:
        target = args.data_dir / "setoa.dat"
        shutil.copy2(target, target.with_name(f"setoa.dat.{stamp:%Y%m%d-%H%M%S}.bak"))
        shutil.copy2(out / "setoa.dat", target)
        print(f"copied the batch into {args.data_dir} (backup kept)")
    print(f"done: {out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
